const headers = { 'Cache-Control': 'no-store' };
const MAX_BODY = 192 * 1024;
const invalid = () => Response.json({ error: 'Use a supported repair action with its exact named text fields', code: 'invalid_request' }, { status: 400, headers });

export async function GET(request: Request) {
  if (!isLocalRequest(request)) return Response.json({ error: 'Local source repair is available on loopback only', code: 'nonlocal' }, { status: 403, headers });
  const url = new URL(request.url);
  if ([...url.searchParams.keys()].some((key) => key !== 'id') || url.searchParams.getAll('id').length !== 1) return invalid();
  const id = url.searchParams.get('id') || '';
  if (!/^[a-f0-9]{24}$/.test(id)) return invalid();
  try {
    const result = await runLocalFileRepair({ action: 'source', file_id: id });
    return Response.json(result.data || { error: result.error, code: result.code }, { status: result.status, headers });
  } catch {
    return Response.json({ error: 'Refresh the local inventory before reviewing this source', code: 'index_unavailable' }, { status: 409, headers });
  }
}

export async function POST(request: Request) {
  const denied = await mutationAuthError(request);
  if (denied) return denied;
  let value: unknown;
  try {
    if (Number(request.headers.get('content-length') || 0) > MAX_BODY) return invalid();
    if (!request.body) return invalid();
    const reader = request.body.getReader();
    const chunks: Uint8Array[] = [];
    let size = 0;
    while (true) {
      const item = await reader.read();
      if (item.done) break;
      size += item.value.byteLength;
      if (size > MAX_BODY) { await reader.cancel(); return invalid(); }
      chunks.push(item.value);
    }
    value = JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch { return invalid(); }
  if (!value || typeof value !== 'object' || Array.isArray(value)) return invalid();
  const fields = value as Record<string, unknown>;
  const keys: Record<string, string[]> = {
    preview: ['action', 'file_id', 'expected_sha256', 'new_content'],
    apply: ['action', 'repair_id', 'proposal_sha256'],
    undo: ['action', 'repair_id'],
  };
  if (typeof fields.action !== 'string' || !Object.hasOwn(keys, fields.action)) return invalid();
  const expected = keys[fields.action];
  if (Object.keys(fields).length !== expected.length || expected.some((key) => typeof fields[key] !== 'string')) return invalid();
  if (fields.action === 'preview' && (!/^[a-f0-9]{24}$/.test(String(fields.file_id)) || !/^[a-f0-9]{64}$/.test(String(fields.expected_sha256)) || Buffer.byteLength(String(fields.new_content)) > 65536)) return invalid();
  if (fields.action !== 'preview' && !/^[a-f0-9]{32}$/.test(String(fields.repair_id))) return invalid();
  if (fields.action === 'apply' && !/^[a-f0-9]{64}$/.test(String(fields.proposal_sha256))) return invalid();
  try {
    const result = await runLocalFileRepair(value as RepairRequest);
    return Response.json(result.data || { error: result.error, code: result.code }, { status: result.status, headers });
  } catch {
    return Response.json({ error: 'Local source repair is unavailable; inspect the inventory and private recovery state', code: 'repair_unavailable' }, { status: 409, headers });
  }
}

import path from 'node:path';
import type { LocalRepairPreview, LocalRepairResult, LocalRepairSource } from '@/lib/types/local-files';
import { localFileIndex } from './local-files';
import { ensureLocalDataDir, foundationRoot, localDataDir } from './paths';

export type RepairRequest =
  | { action: 'preview'; file_id: string; expected_sha256: string; new_content: string }
  | { action: 'apply'; repair_id: string; proposal_sha256: string }
  | { action: 'undo'; repair_id: string };
type SourceRequest = { action: 'source'; file_id: string };
type RepairResponse = { status: number; data?: LocalRepairSource | LocalRepairPreview | LocalRepairResult; error?: string; code?: string };

export async function runLocalFileRepair(input: RepairRequest | SourceRequest): Promise<RepairResponse> {
  // Browser requests select only indexed IDs; root and state paths stay server-owned.
  await localFileIndex();
  ensureLocalDataDir();
  const args = [path.join(foundationRoot(), 'tools', 'local_file_repairs.py'), '--index', path.join(localDataDir(), 'local-files.json'), '--state', localDataDir()];
  return new Promise((resolve) => {
    const child = spawn('python3', args, { stdio: ['pipe', 'pipe', 'ignore'], env: { ...process.env, PYTHONDONTWRITEBYTECODE: '1' } });
    let output = '';
    let settled = false;
    const finish = (value: RepairResponse) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(value);
    };
    const failed = () => finish({ status: 409, code: 'repair_unavailable', error: 'Repair operation interrupted; inspect source and private recovery state before continuing' });
    const timer = setTimeout(() => { child.kill('SIGKILL'); failed(); }, 120_000);
    child.stdout.on('data', (chunk: Buffer) => {
      output += chunk.toString();
      if (Buffer.byteLength(output) > 512 * 1024) { child.kill('SIGKILL'); failed(); }
    });
    child.on('error', failed);
    child.stdin.on('error', failed);
    child.on('close', (code) => {
      if (code !== 0) { failed(); return; }
      try {
        const result = JSON.parse(output) as RepairResponse;
        if (![200, 400, 409].includes(result.status) || (!result.data && !result.error)) { failed(); return; }
        finish(result);
      } catch { failed(); }
    });
    child.stdin.end(JSON.stringify(input));
  });
}

def git(repo, *args):

class RepairTests(unittest.TestCase):

        self.repo = self.home / '.agents'
        self.repo.mkdir()
        self.rules = self.repo / 'instructions'
        self.rules.mkdir()

        self.before = 'Always keep reviewed source evidence.\n'
        self.after = 'Always preserve reviewed source evidence.\n'
        self.file.write_text(self.before)
        self.file.chmod(0o640)
        (self.rules / 'other.md').write_text('Other original\n')
        (self.rules / 'dirty.md').write_text('Dirty original\n')
        git(self.repo, 'init', '-b', 'main')
        git(self.repo, 'config', 'user.email', 'fixture@example.invalid')
        git(self.repo, 'config', 'user.name', 'Fixture')
        git(self.repo, 'add', '.')
        git(self.repo, 'commit', '-m', 'Fixture baseline')

        subprocess.run(['git', 'init', '--bare', str(self.remote)], check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        git(self.repo, 'remote', 'add', 'origin', str(self.remote))
        git(self.repo, 'push', '-u', 'origin', 'main')
        (self.home / 'Backups').mkdir()

        self.state.mkdir()
