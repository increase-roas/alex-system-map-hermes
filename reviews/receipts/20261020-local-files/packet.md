import hashlib
import json
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch
import local_files as lf


class LocalFilesTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.home = Path(self.tmp.name)
        self.root = self.home / '.agents'
        self.root.mkdir()

    def tearDown(self):
        self.tmp.cleanup()

    def put(self, name, text):
        p = self.root / name
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(text)
        return p

    def scan(self, **kw):
        return lf.scan(self.home, roots=[(self.root, 'fixture')], **kw)

    def test_exact_copies_links_generated_and_evidence(self):
        a = self.put('instructions/a.md', 'Always use Hindsight as the only memory layer.\nNever send customer emails.\n')
        self.put('instructions/b.md', a.read_text())
        (self.root / 'linked.md').symlink_to(a)
        self.put('generated.md', '<!-- GENERATED\nSources: instructions/a.md\n-->\nAlways use local files as the only memory layer.\nAlways send customer emails.\n')
        idx = self.scan()
        self.assertEqual({'same_file', 'identical_copy'}, {g['kind'] for g in idx['duplicates']})
        self.assertTrue(any(r['kind'] == 'generated_from' for r in idx['references']))
        self.assertTrue(any(f['kind'] == 'repeated_rule' for f in idx['findings']))
        candidates = [f for f in idx['findings'] if f['kind'] == 'conflict_candidate']
        self.assertTrue(candidates)
        self.assertTrue(all(f['confidence'] == 'candidate' and len(f['evidence']) >= 2 and all(e['line'] > 0 for e in f['evidence']) for f in candidates))

    def test_content_changed_missing_and_unknown_id(self):
        p = self.put('rule.md', 'Always keep the original.\n')
        idx = self.scan()
        row = idx['files'][0]
        p.write_text('Changed contents\n')
        self.assertTrue(lf.content(idx, row['id'])['changed_since_scan'])
        p.unlink()
        self.assertIn('error', lf.content(idx, row['id']))
        with self.assertRaises(KeyError):
            lf.content(idx, '../../etc/passwd')

    def test_secret_screening_recursive_config_and_evidence(self):
        raw = '{"apiKey":"arbitrary-value-should-not-leak","nested":{"password":"odd credential"},"commands":["curl -H Authorization: Bearer myprivatevalue https://example.com"],"safe":"visible"}'
        self.put('settings.json', raw)
        self.put('rules.md', 'export API_KEY=hidden-value\n-----BEGIN PRIVATE KEY-----\nprivatebody\n-----END PRIVATE KEY-----\nNever send token=abc123 customer emails.\n')
        idx = self.scan()
        combined = json.dumps(idx) + ''.join(lf.content(idx, f['id'])['content'] for f in idx['files'])
        for secret in ['arbitrary-value-should-not-leak', 'odd credential', 'myprivatevalue', 'hidden-value', 'privatebody', 'abc123']:
            self.assertNotIn(secret, combined)
        self.assertIn('visible', combined)
        self.assertTrue(all(lf.content(idx, f['id'])['redacted'] for f in idx['files']))
        self.assertEqual(hashlib.sha256(raw.encode()).hexdigest(), next(f['sha256'] for f in idx['files'] if f['path'].endswith('settings.json')))

    def test_boundary_alias_cycle_broken_exclusions_and_limits(self):
        self.put('rules/one.md', 'one')
        (self.root / 'alias').symlink_to(self.root / 'rules', target_is_directory=True)
        (self.root / 'rules' / 'cycle').symlink_to(self.root, target_is_directory=True)
        outside = self.home / 'outside.md'
        outside.write_text('outside private')
        (self.root / 'escape.md').symlink_to(outside)
        (self.root / 'broken.md').symlink_to(self.root / 'absent.md')
        self.put('.env', 'TOP_SECRET=value')
        self.put('auth.json', '{"secret":"private"}')
        self.put('node_modules/no.md', 'dependency')
        self.put('big.md', 'x' * 300)
        idx = self.scan(max_bytes=100)
        self.assertFalse(any(f['path'].endswith(('escape.md', '.env', 'auth.json', 'no.md')) for f in idx['files']))
        self.assertTrue(any(f['path'].endswith('alias/one.md') for f in idx['files']))
        self.assertTrue(any(f['state'] == 'missing' for f in idx['files']))
        self.assertTrue(any(f['state'] == 'too_large' for f in idx['files']))
        self.assertTrue(idx['coverage']['truncated'])
        one = self.scan(max_files=1)
        self.assertEqual(1, len(one['files']))
        self.assertTrue(one['coverage']['truncated'])

    def test_multiline_secret_sections_and_line_proof(self):
        text = '[env]\nUNUSUAL_NAME = "arbitrary-hidden"\n[normal]\nmode = "visible"\n'
        screened, redacted = lf.screen(text)
        self.assertNotIn('arbitrary-hidden', screened)
        self.assertIn('visible', screened)
        self.assertEqual(text.count('\n'), screened.count('\n'))
        text = 'env:\n  ODD_NAME: hidden-yaml-value\nsafe: visible\n'
        screened, _ = lf.screen(text)
        self.assertNotIn('hidden-yaml-value', screened)
        self.assertIn('visible', screened)

    def test_command_argument_containers_withhold_arbitrary_credentials(self):
        secret = 'fixture-opaque-value-723'
        fixtures = [
            '{"command":"bridge","args":["--api-key","' + secret + '"]}',
            'command = "bridge"\nargs = ["--api-key", "' + secret + '"]\n',
            'command = "bridge"\nargs = [\n "--api-key",\n "' + secret + '",\n]\nsafe = "visible"\n',
            'command: bridge\nargs:\n  - --api-key\n  - ' + secret + '\nsafe: visible\n',
            'args:\n- --api-key\n- ' + secret + '\nsafe: visible\n',
            'command: bridge\nargs: ["--api-key", "' + secret + '"]\n',
        ]
        for text in fixtures:
            with self.subTest(format=text.splitlines()[0][:15]):
                screened, redacted = lf.screen(text)
                self.assertNotIn(secret, screened)
                self.assertTrue(redacted)
                self.assertEqual(text.count('\n'), screened.count('\n'))
        self.put('settings.json', fixtures[0])
        idx = self.scan()
        self.assertNotIn(secret, json.dumps(idx) + lf.content(idx, idx['files'][0]['id'])['content'])

    def test_alias_to_excluded_target_is_never_read(self):
        target = self.put('auth.json', '{"safe":"fixture-private-auth"}')
        (self.root / 'harmless.md').symlink_to(target)
        idx = self.scan()
        self.assertEqual([], idx['files'])
        self.assertNotIn('fixture-private-auth', json.dumps(idx))
        # A stale stored row must also be refused by the content boundary.
        forged = {'files': [{'id': 'old-id', 'path': str(self.root / 'harmless.md'), 'real_path': str(target), 'sha256': None}], 'coverage': idx['coverage']}
        self.assertIn('error', lf.content(forged, 'old-id'))

    def test_fair_roots_prioritize_harness_rules_with_hard_aggregate_budget(self):

        skills.mkdir()
        for i in range(100):
            (skills / ('skill-%03d.md' % i)).write_text('x' * 262144)
        codex = self.home / '.codex'
        codex.mkdir()
        (codex / 'AGENTS.md').write_text('Always keep Codex rules visible.\n')
        (codex / 'config.toml').write_text('mode = "fixture"\n')
        claude = self.home / '.claude'
        claude.mkdir()
        (claude / 'CLAUDE.md').write_text('Always keep Claude rules visible.\n')
        alias = self.home / '.cursor'
        alias.symlink_to(claude, target_is_directory=True)
        roots = [(skills, 'large skills'), (codex, 'Codex'), (claude, 'Claude'), (alias, 'Cursor alias')]
        actual_bytes = 0
        original = lf.read_source
        def measured(*args, **kwargs):
            nonlocal actual_bytes

import fs from 'node:fs';
import path from 'node:path';
import type { LocalFileContent, LocalFileIndex } from '@/lib/types/local-files';
import { ensureLocalDataDir, foundationRoot, localDataDir } from './paths';

let inFlight: Promise<LocalFileIndex> | null = null;
const TTL_MS = 60_000;
const indexPath = () => path.join(localDataDir(), 'local-files.json');

function run(command: 'scan' | 'content', id?: string): Promise<unknown> {
  ensureLocalDataDir();
  const args = [path.join(foundationRoot(), 'tools', 'local_files.py'), command, '--index', indexPath()];
  if (id) args.push('--id', id);
  return new Promise((resolve, reject) => {
    const child = spawn('python3', args, { stdio: ['ignore', 'pipe', 'ignore'] });
    let output = '';
    let settled = false;
    const finish = (error?: Error, value?: unknown) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      if (error) reject(error); else resolve(value);
    };
    const timer = setTimeout(() => {
      child.kill('SIGKILL');
      finish(new Error('Local file scan exceeded its time limit'));
    }, 60_000);
    child.stdout.on('data', (chunk: Buffer) => {
      output += chunk.toString();
      if (output.length > 24 * 1024 * 1024) {
        child.kill('SIGKILL');
        finish(new Error('Local file response exceeded its size limit'));
      }
    });
    child.on('error', () => finish(new Error('Unable to start local file reader')));
    child.on('close', (code) => {
      if (code !== 0) { finish(new Error('Local file reader failed')); return; }
      try { finish(undefined, JSON.parse(output)); }
      catch { finish(new Error('Invalid local file reader response')); }
    });
  });
}

export async function localFileIndex(refresh = false): Promise<LocalFileIndex> {
  if (inFlight) return inFlight;
  if (!refresh) {
    try {
      const value = JSON.parse(fs.readFileSync(indexPath(), 'utf8')) as LocalFileIndex;
      const age = Date.now() - Date.parse(value.scanned_at);
      const scannerMtime = fs.statSync(path.join(foundationRoot(), 'tools', 'local_files.py')).mtimeMs;
      if (value.schema_version === 1 && age >= 0 && age < TTL_MS && fs.statSync(indexPath()).mtimeMs >= scannerMtime) return value;
    } catch { /* Missing or invalid snapshot gets rebuilt from selected roots. */ }
  }
  inFlight = run('scan').then((value) => value as LocalFileIndex).finally(() => { inFlight = null; });
  return inFlight;
}

export async function localFileContent(id: string): Promise<LocalFileContent | null> {
  if (!/^[a-f0-9]{24}$/.test(id)) return null;
  const index = await localFileIndex();
  if (!index.files.some((file) => file.id === id)) return null;
  return await run('content', id) as LocalFileContent;
}

export async function GET(request: Request) {
  if (!isLocalRequest(request)) return Response.json({ error: 'Local files are available on loopback only' }, { status: 403, headers });
  const url = new URL(request.url);
  if ([...url.searchParams.keys()].some((key) => key !== 'id')) {
    return Response.json({ error: 'Only indexed file IDs are accepted' }, { status: 400, headers });
  }
  try {
    if (url.searchParams.has('id')) {
      const value = await localFileContent(url.searchParams.get('id') || '');
      return value ? Response.json(value, { headers }) : Response.json({ error: 'Unknown indexed file ID' }, { status: 404, headers });
    }
    return Response.json(await localFileIndex(), { headers });
  } catch {
    return Response.json({ error: 'Unable to read the bounded local file inventory' }, { status: 500, headers });
  }
}

export async function POST(request: Request) {
  const denied = await mutationAuthError(request);
  if (denied) return denied;
  try {
    return Response.json(await localFileIndex(true), { headers });
  } catch {
    return Response.json({ error: 'Unable to refresh the bounded local file inventory' }, { status: 500, headers });
  }
}

function membershipByComponent(bundle: CatalogBundle) {
  return new Map(bundle.overview_memberships.map((item) => [item.component_id, item]));

  return bundle.overview_memberships.filter((item) => item.parent_id === parentId).length;

function componentLayer(bundle: CatalogBundle, componentId: string): string | null {

/** Map a component to a layer node or expandable group node on the overview canvas. */
function overviewAnchor(bundle: CatalogBundle, componentId: string): string | null {
  const membership = membershipByComponent(bundle).get(componentId);
  const layerId = membership?.layer_id ?? componentLayer(bundle, componentId);
  if (!layerId) {
    return null;
  }
  if (membership?.parent_id) {
    return `group:${membership.parent_id}`;
  }
  if (childCount(bundle, componentId) > 0) {
    return `group:${componentId}`;
  }
  return layerId;
}

/** Per-layer breakdown: total = groupRoots + groupedChildren + standalone. Every membership is counted once. */
export type LayerBreakdown = {
  total: number;
  groups: { componentId: string; childCount: number }[];
  groupedChildren: number;
  standalone: number;
};

export function layerBreakdown(bundle: CatalogBundle): Map<string, LayerBreakdown> {
  const children = new Map<string, number>();
  for (const member of bundle.overview_memberships) {
    if (member.parent_id) {
      children.set(member.parent_id, (children.get(member.parent_id) ?? 0) + 1);
    }
  }
  const out = new Map<string, LayerBreakdown>();
  for (const layer of bundle.data.layers) {
    out.set(layer.id, { total: 0, groups: [], groupedChildren: 0, standalone: 0 });
  }
  for (const member of bundle.overview_memberships) {
    const entry = out.get(member.layer_id);
    if (!entry) {
      continue;
    }
    entry.total += 1;
    if (member.parent_id) {
      entry.groupedChildren += 1;
    } else if (children.has(member.component_id)) {
      entry.groups.push({ componentId: member.component_id, childCount: children.get(member.component_id) ?? 0 });
    } else {
      entry.standalone += 1;
    }
  }
  for (const entry of out.values()) {
    entry.groups.sort((a, b) => b.childCount - a.childCount);
  }
  return out;
}

export const OVERVIEW_LAYOUT = {
  layerX: 40,
  layerWidth: 240,
  groupX: 320,
  groupWidth: 260,
  groupHeight: 156,
  gapX: 24,
  gapY: 20,
  groupsPerRow: 4,
  rowGap: 48,
};

export function buildOverviewGraph(bundle: CatalogBundle): { nodes: Node[]; edges: Edge[] } {
  const breakdown = layerBreakdown(bundle);
  const sortedLayers = [...bundle.data.layers].sort((a, b) => a.order - b.order);
  const L = OVERVIEW_LAYOUT;
  const nodes: Node[] = [];
  const groupsByLayer = new Map<string, string[]>();

  let y = 0;
  for (const layer of sortedLayers) {
    const info = breakdown.get(layer.id) ?? { total: 0, groups: [], groupedChildren: 0, standalone: 0 };
    const rows = Math.max(1, Math.ceil(info.groups.length / L.groupsPerRow));
    const bandHeight = rows * L.groupHeight + (rows - 1) * L.gapY;
    nodes.push({
      id: layer.id,
      type: 'layerNode',
      position: { x: L.layerX, y },
      data: {
        label: layer.name,
        count: info.total,
        groupCount: info.groups.length,
        groupedChildren: info.groupedChildren,
        standalone: info.standalone,
        layerId: layer.id,
        layerOrder: layer.order,
      },
    });
    groupsByLayer.set(layer.id, info.groups.map((g) => g.componentId));
    info.groups.forEach((group, index) => {
      const component = bundle.data.components.find((item) => item.id === group.componentId);
      nodes.push({
        id: `group:${group.componentId}`,
        type: 'groupNode',
        position: {
          x: L.groupX + (index % L.groupsPerRow) * (L.groupWidth + L.gapX),
          y: y + Math.floor(index / L.groupsPerRow) * (L.groupHeight + L.gapY),
        },
        data: {
          ...componentData(component),
          isGroup: true,
          childCount: group.childCount,
          layerTotal: info.total,
          layerId: layer.id,
          componentId: group.componentId,
        },
      });
    });
    y += bandHeight + L.rowGap;
  }

  const edges: Edge[] = [];
  for (let i = 0; i < sortedLayers.length - 1; i += 1) {
    const from = sortedLayers[i];
    const to = sortedLayers[i + 1];
    edges.push({
      id: `layer-flow-${from.id}-${to.id}`,
      source: from.id,
      target: to.id,
      label: 'layer order',
      data: { kind: 'layer_navigation' },
    });
  }
  for (const layer of sortedLayers) {
    for (const componentId of groupsByLayer.get(layer.id) ?? []) {
      edges.push({
        id: `layer-group-${layer.id}-${componentId}`,
        source: layer.id,
        target: `group:${componentId}`,
        label: 'navigation group',
        data: { kind: 'membership' },
      });
    }
  }

  const seen = new Set<string>();
  for (const rel of bundle.data.relationships) {
    const source = overviewAnchor(bundle, rel.source_id);
    const target = overviewAnchor(bundle, rel.target_id);
    if (!source || !target || source === target) {
      continue;
    }
    const key = `${source}->${target}:${rel.label}`;
    if (seen.has(key)) {
      continue;
    }
    seen.add(key);
    edges.push({
      id: rel.id,
      source,
      target,
      label: rel.label,
      data: { relationshipId: rel.id, kind: 'configured' },
    });
    if (edges.length >= 32) {
      break;
    }
  }
  return { nodes, edges };
}

export function buildSubsystemGraph(
  bundle: CatalogBundle,
  layerId: string,
  expandedGroupId: string | null,
  limit = 120,
): { nodes: Node[]; edges: Edge[] } {
  const inLayer = bundle.overview_memberships.filter((item) => item.layer_id === layerId);
  const roots = inLayer.filter((item) => !item.parent_id);
  const groups = roots.slice(0, 40);
  const nodes: Node[] = [];
  const memberIds = new Set<string>();

  if (!expandedGroupId) {
    for (const [index, group] of groups.entries()) {
      const childCount = inLayer.filter((item) => item.parent_id === group.component_id).length;
      const self = bundle.data.components.find((c) => c.id === group.component_id);
      nodes.push({
        id: group.component_id,
        type: 'componentNode',
        position: { x: (index % 6) * 200, y: Math.floor(index / 6) * 110 },
        data: {
          ...componentData(self),
          isGroup: childCount > 0,
          childCount,
        },
      });
      memberIds.add(group.component_id);
    }
  } else {
    const children = inLayer
      .filter((item) => item.parent_id === expandedGroupId || item.component_id === expandedGroupId)
      .slice(0, limit);
    for (const [index, member] of children.entries()) {
      const component = bundle.data.components.find((item) => item.id === member.component_id);
      nodes.push({
        id: member.component_id,
        type: 'componentNode',
        position: { x: (index % 8) * 180, y: Math.floor(index / 8) * 100 },
        data: componentData(component),
      });
      memberIds.add(member.component_id);
    }
  }

  const edges: Edge[] = bundle.data.relationships
    .filter((rel) => memberIds.has(rel.source_id) && memberIds.has(rel.target_id))
    .map((rel) => ({
      id: rel.id,
      source: rel.source_id,
      target: rel.target_id,
      label: rel.label,
      data: { relationshipId: rel.id, kind: 'configured' },
    }));

  return { nodes, edges };
}

export function componentData(component?: CatalogComponent) {

    evidence_level: component.evidence_level,

    componentId: component.id,
  };
}

export function searchComponents(
  bundle: CatalogBundle,
  query: string,
): CatalogComponent[] {
  const q = query.trim().toLowerCase();
  if (!q) {
    return [];
  }
  const exact = bundle.data.components.filter((item) => item.id.toLowerCase() === q);
  if (exact.length) {
    return exact;
  }
  return bundle.data.components
    .filter((item) => {
      const name = (item.name ?? '').toLowerCase();
      const id = (item.id ?? '').toLowerCase();
      const aliases = item.aliases ?? [];
      return (
        name.includes(q) ||
        id.includes(q) ||
        aliases.some((alias) => String(alias).toLowerCase().includes(q))
      );
    })
    .slice(0, 40);
}
